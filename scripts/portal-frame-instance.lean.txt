theorem structuralWitness : StructuralPremises model actions supports cutParameter := by
  constructor
  · intro segment
    fin_cases segment <;> simp [segmentOwnershipValid, locationBelongs, model]
  · intro segment
    fin_cases segment <;> norm_num [model, locationParameter]
  · intro member t ht0 ht1
    refine ⟨member, ?_⟩
    fin_cases member <;> simpa [model, locationParameter] using And.intro ht0 ht1
  · intro a b hne hmember
    exact False.elim (hne hmember)
  · intro member left right hl hr _ _ hp
    cases left with
    | interior owner cut => exact Fin.elim0 cut
    | endSide lhs =>
      cases right with
      | interior owner cut => exact Fin.elim0 cut
      | endSide rhs =>
        rcases lhs with ⟨lm, le⟩
        rcases rhs with ⟨rm, re⟩
        cases le <;> cases re <;> simp_all [locationBelongs, locationParameter]
  · intro member
    fin_cases member
    · exact ⟨⟨0, Or.inl rfl⟩, ⟨0, Or.inr rfl⟩⟩
    · exact ⟨⟨1, Or.inl rfl⟩, ⟨1, Or.inr rfl⟩⟩
    · exact ⟨⟨2, Or.inl rfl⟩, ⟨2, Or.inr rfl⟩⟩
  · intro member cut
    exact Fin.elim0 cut
  · intro action
    exact Fin.elim0 action
  · intro support
    fin_cases support
    · exact ⟨0, Or.inl rfl⟩
    · exact ⟨2, Or.inr rfl⟩

theorem portalContract : ExactContract model actions supports cutParameter
    (memberDistance nodePosition drawingStart drawingEnd) sourceEA sourceEI := by
  exact (compose_exact_contract model actions supports cutParameter nodePosition drawingStart drawingEnd
    sourceEA sourceEI segmentStart segmentEnd supportAngles structuralWitness geometryWitness
    (by
      intro m h
      have hx := congrArg Prod.fst h
      have hy := congrArg Prod.snd h
      fin_cases m <;> norm_num [nodePosition, drawingStart, drawingEnd] at hx hy)
    (by intro m; fin_cases m <;> norm_num [sourceEA])
    (by intro m; fin_cases m <;> norm_num [sourceEI])
    supportAxesWitness springInputsWitness).1

-- Local recovery is checked for any supplied model coordinate vector. This
-- statement does not assert that the browser vector satisfies equilibrium.
theorem portalLocalRecovery (q : Coordinate model → ℝ) (s : Fin 3) :
    Nonempty (LocalFrameFields.SegmentSolution (segmentDofs model q s)
      (model.segmentLength s) (model.segmentEA s) (model.segmentEI s)
      ((model.member (model.segmentMember s)).orientation.sine * model.segmentVerticalStart s)
      ((model.member (model.segmentMember s)).orientation.sine * model.segmentVerticalEnd s)
      ((model.member (model.segmentMember s)).orientation.cosine * model.segmentVerticalStart s)
      ((model.member (model.segmentMember s)).orientation.cosine * model.segmentVerticalEnd s)) :=
  recovered_segment_exists model q s positivePortalSegments
